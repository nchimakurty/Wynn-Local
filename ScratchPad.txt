Test a single key end-to-end (Aggregation Builder)

[
  // Stage 1: limit to ONE key so you can test safely
  {
    $match: {
      venue_id: { $exists: true, $ne: null, $ne: "" },
      venue_key: { $exists: true, $ne: null, $ne: "", $eq: "ksdflkjsd" } // <-- change this to the key you want to test
    }
  },

  // Stage 2: only where venue_id == venue_key
  {
    $match: { $expr: { $eq: ["$venue_id", "$venue_key"] } }
  },

  // Stage 3: apply mapping (switch keeps everything else unchanged)
  {
    $set: {
      venue_key: {
        $switch: {
          branches: [
            { case: { $eq: ["$venue_key", "ksdflkjsd"] }, then: "id79854" },
            { case: { $eq: ["$venue_key", "yiruytv,mnf"] }, then: "id09320492" } // duplicate resolved to this
          ],
          default: "$venue_key"
        }
      }
    }
  },

  // Stage 4: write back
  {
    $merge: {
      into: "IB_Ops_DiningReservations",
      on: "_id",
      whenMatched: "merge",
      whenNotMatched: "discard"
    }
  }
]
